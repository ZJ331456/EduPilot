   改进后：使用 constants.js 统一管理

                         使用指南
================================================================================

新功能开发时，请使用工具函数：

// ✅ 推荐做法
import { showSuccess, formatDate, safeGet } from '@/utils'

try {
  const data = await api.someMethod()
  const result = safeGet(data, 'data.result', {})
  showSuccess('操作成功')
} catch (error) {
  showError(error, '操作失败')
}

// ❌ 避免做法
import { ElMessage } from 'element-plus'
ElMessage.success('成功')  // 不统一
const result = data.data.result  // 不安全
